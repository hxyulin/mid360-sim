cmake_minimum_required(VERSION 3.16)
project(mid360_gazebo VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_library(mid360-gazebo SHARED src/Mid360.cc)
target_include_directories(mid360-gazebo PRIVATE include)
target_link_libraries(mid360-gazebo PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
target_compile_options(mid360-gazebo PRIVATE -Wall -Wextra -Werror)
install(TARGETS mid360-gazebo LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES tests/enclosure.sdf DESTINATION ${CMAKE_INSTALL_DATADIR}/mid360-gazebo)
install(FILES ../../LICENSE-MIT ../../LICENSE-APACHE ../../NOTICE.md
  DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mid360-gazebo)
include(CTest)
if(BUILD_TESTING)
add_executable(pattern-test tests/pattern.cc)
target_include_directories(pattern-test PRIVATE include)
add_test(NAME pattern COMMAND pattern-test)
add_library(replay-test SHARED tests/Replay.cc)
target_link_libraries(replay-test PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_options(replay-test PRIVATE -Wall -Wextra -Werror)
add_executable(native-test tests/native.cc)
target_include_directories(native-test PRIVATE include)
target_link_libraries(native-test PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
target_compile_options(native-test PRIVATE -Wall -Wextra -Werror)
endif()
